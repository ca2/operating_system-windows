cmake_minimum_required(VERSION 3.20)
project(windows_xp_adapters_test LANGUAGES CXX)
enable_testing()
foreach(mode IN ITEMS native xp)
    add_executable(printf_format_${mode} printf_format.cpp)
    target_include_directories(printf_format_${mode} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../../../../source/app")
    target_compile_features(printf_format_${mode} PRIVATE cxx_std_11)
    if(mode STREQUAL "xp")
        target_compile_definitions(printf_format_${mode} PRIVATE __REACTOS__)
    endif()
    add_test(NAME printf_format_${mode} COMMAND printf_format_${mode})
endforeach()
add_executable(xp_ffmpeg_utf8 ffmpeg_utf8.cpp)
target_compile_definitions(xp_ffmpeg_utf8 PRIVATE __REACTOS__)
add_test(NAME xp_ffmpeg_utf8 COMMAND xp_ffmpeg_utf8)
foreach(mode IN ITEMS native fallback)
    add_executable(xp_adapters_${mode} adapters.cpp)
    target_include_directories(xp_adapters_${mode} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../../../../source/app")
    target_link_libraries(xp_adapters_${mode} PRIVATE dbghelp advapi32)
    if(mode STREQUAL "fallback")
        target_compile_definitions(xp_adapters_${mode} PRIVATE FORCE_XP_FALLBACK)
    endif()
    add_test(NAME xp_adapters_${mode} COMMAND xp_adapters_${mode})
endforeach()
