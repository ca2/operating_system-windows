cmake_minimum_required(VERSION 3.20)
project(nano_http_mbedtls_with_winsock LANGUAGES CXX)
if(NOT TARGET ca2::mbedtls)
    get_filename_component(wrapper "${CMAKE_CURRENT_SOURCE_DIR}/../../../port/common/mbedtls" ABSOLUTE)
    add_subdirectory("${wrapper}" "${CMAKE_BINARY_DIR}/tls")
endif()
# Embed a pinned root bundle: XP's obsolete trust store is not used.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/ca-bundle.pem" ca_hex HEX)
string(REGEX REPLACE "(..)" "0x\\1," ca_hex "${ca_hex}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ca_bundle.h" "static const unsigned char ca_bundle[] = {${ca_hex}0};\n")
add_library(nano_http_mbedtls_with_winsock_transport STATIC transport.cpp)
target_compile_features(nano_http_mbedtls_with_winsock_transport PUBLIC cxx_std_17)
target_include_directories(nano_http_mbedtls_with_winsock_transport PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(nano_http_mbedtls_with_winsock_transport PUBLIC ca2::mbedtls ws2_32)
add_library(${PROJECT_NAME} SHARED http.cpp _factory.cpp http.h platform.h _.h)
target_compile_definitions(${PROJECT_NAME} PRIVATE _nano_http_mbedtls_with_winsock_project)
target_link_libraries(${PROJECT_NAME} PRIVATE acme nano_http_mbedtls_with_winsock_transport)
